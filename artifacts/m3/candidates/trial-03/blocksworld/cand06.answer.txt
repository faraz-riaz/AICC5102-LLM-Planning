```python
from heuristics.heuristic_base import Heuristic


class blocksworldLLMHeuristic(Heuristic):
    """
    A domain-dependent heuristic for the blocksworld domain.

    # Summary
    This heuristic estimates the number of actions required to reach a goal state.
    It identifies which blocks are already "well-placed" (in their final correct position 
    with no need to be moved later) and calculates the cost to move all other blocks 
    to their target positions. It effectively avoids local minima by correctly estimating 
    whether a block's target position is currently available, incentivizing the search 
    to clear blocks that are in the way.

    # Assumptions
    - Goals primarily consist of `on`, `on-table`, and `clear` predicates.
    - A block that is well-placed does not need to be moved.
    - Moving a block that is not well-placed takes 1 to 4 actions depending on its 
      current state (held, on table, on another block) and whether its destination is ready.

    # Heuristic Initialization
    - Extract `on`, `on-table`, and `clear` conditions from the goals.
    - Map each block to its target block (`goal_on`) and vice versa (`goal_on_inv`).
    - Identify all blocks present in the problem through the initial state and goals.

    # Step-By-Step Thinking for Computing Heuristic
    1. Parse the current state to find where each block is (`state_on`, `state_on_table`, 
       `state_clear`, `state_holding`).
    2. Determine the set of `well_placed` blocks iteratively. A block is well-placed if:
       - It is on the table and does not need to be on another block.
       - Or it is on a well-placed block `c`, it belongs on `c` (or has no specific target),
         `c` does not need to be clear, and no other block needs to be on `c`.
    3. For each block not well-placed, determine its cost:
       - If it is currently held: cost is 1 if its target is ready, 3 otherwise.
       - If it is on the table: cost is 2 (it just needs to wait until target is ready to 
         be picked up and stacked).
       - If it is on another block: cost is 2 if target is ready, 4 otherwise (since it may 
         need to be moved to the table first to unblock other necessary blocks).
    4. A target for a block `b` is "ready" if `b` needs to go on `c`, and `c` is both 
       well-placed and clear. If `b` belongs on the table or has no specific goal, its 
       target is always considered ready.
    5. The sum of these costs provides a highly accurate gradient for greedy best-first search.
    """

    def __init__(self, task):
        self.goals = task.goals
        
        self.goal_on = {}
        self.goal_on_inv = {}
        self.goal_on_table = set()
        self.goal_clear = set()
        self.all_blocks = set()

        # Extract goals and define targets for blocks
        for goal in self.goals:
            parts = goal[1:-1].split()
            pred = parts[0]
            if pred == "on":
                b, c = parts[1], parts[2]
                self.goal_on[b] = c
                self.goal_on_inv[c] = b
                self.all_blocks.add(b)
                self.all_blocks.add(c)
            elif pred == "on-table":
                self.goal_on_table.add(parts[1])
                self.all_blocks.add(parts[1])
            elif pred == "clear":
                self.goal_clear.add(parts[1])
                self.all_blocks.add(parts[1])

        # Safely capture all blocks from the initial state
        for fact in task.initial_state:
            parts = fact[1:-1].split()
            if parts[0] in ("on", "on-table", "clear", "holding"):
                self.all_blocks.update(parts[1:])

    def __call__(self, node):
        state = node.state
        
        state_on = {}
        state_on_table = set()
        state_clear = set()
        state_holding = None
        
        # Parse the current state dynamically
        for fact in state:
            parts = fact[1:-1].split()
            pred = parts[0]
            if pred == "on":
                state_on[parts[1]] = parts[2]
            elif pred == "on-table":
                state_on_table.add(parts[1])
            elif pred == "clear":
                state_clear.add(parts[1])
            elif pred == "holding":
                state_holding = parts[1]
                
        well_placed = set()
        
        # Base cases: blocks on the table that don't belong on another block
        for b in state_on_table:
            if b not in self.goal_on:
                well_placed.add(b)
                
        # Recursive cases: blocks on well-placed blocks
        changed = True
        while changed:
            changed = False
            for b, c in state_on.items():
                if b not in well_placed and c in well_placed:
                    # 1. b must not be required to be on the table
                    if b in self.goal_on_table:
                        continue
                    # 2. b must be on its required target (if it has one)
                    if b in self.goal_on and self.goal_on[b] != c:
                        continue
                    # 3. c must not be required to be clear in the goal
                    if c in self.goal_clear:
                        continue
                    # 4. c must not be the required target for some other block
                    if c in self.goal_on_inv and self.goal_on_inv[c] != b:
                        continue
                    
                    well_placed.add(b)
                    changed = True
                    
        def is_target_ready(b):
            """Returns True if the block's destination is ready for it to be placed."""
            if b in self.goal_on:
                c = self.goal_on[b]
                return c in well_placed and c in state_clear
            elif b in self.goal_on_table:
                return True
            else:
                # If a block has no specific target, it can always be put on the table.
                return True
                
        total_cost = 0
        
        for b in self.all_blocks:
            # If the block is perfectly in its goal setup, no actions are needed.
            if b in well_placed:
                continue
                
            if b == state_holding:
                if is_target_ready(b):
                    total_cost += 1
                else:
                    # E.g., putdown, pickup, stack
                    total_cost += 3
            elif b in state_on_table:
                # Being on the table implies it doesn't block crucial dependencies,
                # so it just needs a pickup and a stack once its target is ready.
                total_cost += 2
            else:
                # Block is currently resting on another block.
                if is_target_ready(b):
                    total_cost += 2
                else:
                    # Might need to be moved out of the way (unstack, putdown, pickup, stack)
                    total_cost += 4
                    
        return total_cost
```
